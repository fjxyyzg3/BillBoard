# 本机后台运维看板 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现一个本机访问、独立口令保护的 BillBoard 后台运维与账号安全管理界面。

**Architecture:** 使用同一 Next.js 代码库新增 `/ops` 后台路由，后台进程通过环境变量显式启用并由 `npm run ops` 绑定 `127.0.0.1:32768`。`AppEvent` 记录非敏感应用事件；后续 `AppSession` 和 `OpsAuditLog` 支持家庭成员账号改密、应用 session 撤销和最小安全审计，但不修改家庭账本数据。

**2026-05-15 follow-up:** `v1.4.1` keeps direct `OPS_ADMIN_PASSWORD_HASH` support, but env-file guidance from 2026-05-15 prefers `OPS_ADMIN_PASSWORD_HASH_BASE64` so argon2 `$` characters are not expanded by env loaders. Treat the older direct-hash examples below as historical implementation context.

**2026-05-20 entrypoint follow-up:** `v1.11.0` uses nginx root routing for the BillBoard hostname plus frpc remote `3000`. The ops dashboard remains local-only at `127.0.0.1:32768` and must not be published through nginx, frp, or compose.

**2026-05-16 account-security follow-up:** OPS is no longer purely read-only. It now includes limited account-security operations for household member accounts: password reset, automatic revocation of that user's active `AppSession` rows after password reset, single-session revocation, and recent `OpsAuditLog` display. It still must not add/delete users, manage the OPS admin password, modify categories, or modify ledger transactions. See `docs/superpowers/specs/2026-05-09-ops-admin-dashboard-design.md` for the current design.

**Tech Stack:** Next.js App Router, React Server Components, Server Actions, Prisma, PostgreSQL, Auth.js/NextAuth, argon2, Vitest, Playwright.

import { z } from "zod";

const appEventMetadataSchema = z
  .object({
    count: z.number().int().min(0).max(100000).optional(),
    source: z.string().trim().min(1).max(64).optional(),
    transactionType: z.enum(["expense", "income"]).optional(),
  })
  .strict();

export type AppEventMetadataInput = z.input<typeof appEventMetadataSchema>;
export type AppEventMetadata = z.output<typeof appEventMetadataSchema>;

function removeUndefinedValues(metadata: AppEventMetadata) {
  return Object.fromEntries(
    Object.entries(metadata).filter((entry): entry is [string, string | number] => {
      const [, value] = entry;

      return value !== undefined;
    }),
  );
}

export function sanitizeAppEventMetadata(input: unknown) {
  if (input === undefined || input === null) {
    return undefined;
  }

  const result = appEventMetadataSchema.safeParse(input);

  if (!result.success) {
    throw new Error("Unsupported app event metadata");
  }

  const metadata = removeUndefinedValues(result.data);

  return Object.keys(metadata).length > 0 ? metadata : undefined;
}
```

import type { AppEventType } from "@prisma/client";
import { db } from "@/lib/db";
import { sanitizeAppEventMetadata, type AppEventMetadataInput } from "@/lib/ops/event-metadata";

const appEventRetentionDays = 90;
const appEventRetentionMs = appEventRetentionDays * 24 * 60 * 60 * 1000;

export type RecordAppEventInput = {
  eventType: AppEventType;
  householdId?: string | null;
  memberId?: string | null;
  metadata?: AppEventMetadataInput | null;
  now?: Date;
  path?: string | null;
  summary?: string | null;
};

export async function recordAppEvent(input: RecordAppEventInput) {
  try {
    const createdAt = input.now ?? new Date();
    const cutoff = new Date(createdAt.getTime() - appEventRetentionMs);
    const metadata = sanitizeAppEventMetadata(input.metadata);

    await db.$transaction([
      db.appEvent.deleteMany({
        where: { createdAt: { lt: cutoff } },
      }),
      db.appEvent.create({
        data: {
          createdAt,
          eventType: input.eventType,
          householdId: input.householdId ?? null,
          memberId: input.memberId ?? null,
          metadata,
          path: input.path ?? null,
          summary: input.summary ?? null,
        },
      }),
    ]);
  } catch {
    return undefined;
  }
}
```

```ts
import crypto from "node:crypto";
import argon2 from "argon2";

export const opsAdminSessionCookieName = "billboard-ops-session";
const opsAdminSessionMaxAgeMs = 8 * 60 * 60 * 1000;

function signOpsAdminSession(issuedAtMs: number, secret: string) {
  return crypto.createHmac("sha256", secret).update(String(issuedAtMs)).digest("base64url");
}

function safeEqual(left: string, right: string) {
  const leftBuffer = Buffer.from(left);
  const rightBuffer = Buffer.from(right);

  return leftBuffer.length === rightBuffer.length && crypto.timingSafeEqual(leftBuffer, rightBuffer);
}

export async function verifyOpsAdminPassword(password: string) {
  const hash = process.env.OPS_ADMIN_PASSWORD_HASH;

  if (!hash) {
    return false;
  }

  return argon2.verify(hash, password).catch(() => false);
}

export function createOpsAdminSessionToken(secret: string, issuedAt = new Date()) {
  const issuedAtMs = issuedAt.getTime();
  const signature = signOpsAdminSession(issuedAtMs, secret);

  return `${issuedAtMs}.${signature}`;
}

export function verifyOpsAdminSessionToken(token: string | undefined, secret: string, now = new Date()) {
  if (!token) {
    return false;
  }

  const [issuedAtRaw, signature] = token.split(".");
  const issuedAtMs = Number(issuedAtRaw);

  if (!Number.isInteger(issuedAtMs) || !signature) {
    return false;
  }

  if (now.getTime() - issuedAtMs > opsAdminSessionMaxAgeMs) {
    return false;
  }

  return safeEqual(signature, signOpsAdminSession(issuedAtMs, secret));
}
```

import { AppEventType } from "@prisma/client";
import { db } from "@/lib/db";

const shanghaiOffsetMs = 8 * 60 * 60 * 1000;
const dayMs = 24 * 60 * 60 * 1000;

type QueryOpsDashboardInput = {
  now?: Date;
  version: string;
};

function getShanghaiDateParts(date: Date) {
  const shanghai = new Date(date.getTime() + shanghaiOffsetMs);

  return {
    day: shanghai.getUTCDate(),
    month: shanghai.getUTCMonth(),
    year: shanghai.getUTCFullYear(),
  };
}

function getShanghaiStartUtc(date: Date) {
  const parts = getShanghaiDateParts(date);

  return new Date(Date.UTC(parts.year, parts.month, parts.day, -8, 0, 0, 0));
}

function getDayKey(date: Date) {
  const parts = getShanghaiDateParts(date);
  const month = String(parts.month + 1).padStart(2, "0");
  const day = String(parts.day).padStart(2, "0");

  return `${parts.year}-${month}-${day}`;
}

function createEmptyTrend(now: Date) {
  const todayStart = getShanghaiStartUtc(now);

  return Array.from({ length: 7 }, (_, index) => {
    const dayStart = new Date(todayStart.getTime() - (6 - index) * dayMs);

    return {
      dayKey: getDayKey(dayStart),
      keyActions: 0,
      loginSuccess: 0,
      pageViews: 0,
    };
  });
}

function isKeyAction(eventType: AppEventType) {
  return (
    eventType === AppEventType.TRANSACTION_CREATED ||
    eventType === AppEventType.TRANSACTION_UPDATED ||
    eventType === AppEventType.TRANSACTION_DELETED ||
    eventType === AppEventType.IMPORT_CONFIRMED
  );
}

export async function queryOpsDashboard({ now = new Date(), version }: QueryOpsDashboardInput) {
  const todayStart = getShanghaiStartUtc(now);
  const from = new Date(todayStart.getTime() - 6 * dayMs);
  const to = new Date(todayStart.getTime() + dayMs - 1);

  const status = {
    bindAddress: "127.0.0.1:32768",
    environment: process.env.NODE_ENV ?? "development",
    retentionDays: 90,
    version,
  };

  try {
    await db.$queryRaw`SELECT 1`;

    const events = await db.appEvent.findMany({
      where: { createdAt: { gte: from, lte: to } },
      orderBy: { createdAt: "desc" },
      include: { member: { select: { memberName: true } } },
    });
    const trend = createEmptyTrend(now);
    const trendByDay = new Map(trend.map((point) => [point.dayKey, point]));
    const todayEvents = events.filter((event) => event.createdAt >= todayStart);

    for (const event of events) {
      const point = trendByDay.get(getDayKey(event.createdAt));

      if (!point) {
        continue;
      }

      if (event.eventType === AppEventType.LOGIN_SUCCESS) {
        point.loginSuccess += 1;
      } else if (event.eventType === AppEventType.PAGE_VIEW) {
        point.pageViews += 1;
      } else if (isKeyAction(event.eventType)) {
        point.keyActions += 1;
      }
    }

    return {
      database: { ok: true as const },
      recentEvents: events.slice(0, 20).map((event) => ({
        createdAt: event.createdAt,
        eventType: event.eventType,
        memberName: event.member?.memberName ?? null,
        path: event.path,
        summary: event.summary,
      })),
      status,
      today: {
        importConfirmed: todayEvents.filter((event) => event.eventType === AppEventType.IMPORT_CONFIRMED)
          .length,
        loginSuccess: todayEvents.filter((event) => event.eventType === AppEventType.LOGIN_SUCCESS).length,
        pageViews: todayEvents.filter((event) => event.eventType === AppEventType.PAGE_VIEW).length,
        transactionCreated: todayEvents.filter(
          (event) => event.eventType === AppEventType.TRANSACTION_CREATED,
        ).length,
        transactionDeleted: todayEvents.filter(
          (event) => event.eventType === AppEventType.TRANSACTION_DELETED,
        ).length,
        transactionUpdated: todayEvents.filter(
          (event) => event.eventType === AppEventType.TRANSACTION_UPDATED,
        ).length,
      },
      trend,
    };
  } catch (error) {
    return {
      database: {
        message: error instanceof Error ? error.message : "Database connection failed",
        ok: false as const,
      },
      recentEvents: [],
      status,
      today: {
        importConfirmed: 0,
        loginSuccess: 0,
        pageViews: 0,
        transactionCreated: 0,
        transactionDeleted: 0,
        transactionUpdated: 0,
      },
      trend: createEmptyTrend(now),
    };
  }
}

"use server";

import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import {
  createOpsAdminSessionToken,
  opsAdminSessionCookieName,
  verifyOpsAdminPassword,
} from "@/lib/ops/auth";
import { getMessages, parseLocale } from "@/lib/i18n";

export type OpsLoginActionState = {
  status: "idle" | "error";
  message?: string;
};

export async function loginOpsAdmin(
  _previousState: OpsLoginActionState,
  formData: FormData,
): Promise<OpsLoginActionState> {
  const locale = parseLocale(formData.get("locale"));
  const messages = getMessages(locale);
  const password = String(formData.get("password") ?? "");
  const isValid = await verifyOpsAdminPassword(password);

  if (!isValid) {
    return {
      message: messages.ops.invalidPassword,
      status: "error",
    };
  }

  const cookieStore = await cookies();
  cookieStore.set(
    opsAdminSessionCookieName,
    createOpsAdminSessionToken(process.env.AUTH_SECRET ?? "development-auth-secret"),
    {
      httpOnly: true,
      maxAge: 8 * 60 * 60,
      path: "/ops",
      sameSite: "lax",
      secure: process.env.NODE_ENV === "production",
    },
  );

  redirect("/ops");
}

export async function logoutOpsAdmin() {
  const cookieStore = await cookies();
  cookieStore.delete(opsAdminSessionCookieName);

  redirect("/ops");
}
